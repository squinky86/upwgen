cmake_minimum_required(VERSION 3.20)
project(upwgen VERSION 0.1.0 DESCRIPTION "Unicode password generator and authentication test tool" LANGUAGES CXX)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/VERSION" CONTENT "${PROJECT_VERSION}\n")
configure_file(cmake/version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/include/upwgen/version.hpp" @ONLY)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(UPWGEN_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(CTest)
include(GNUInstallDirs)

add_library(upwgen_build_options INTERFACE)
if(MSVC)
    target_compile_options(upwgen_build_options INTERFACE /W4 /WX /utf-8 /sdl /guard:cf)
    target_link_options(upwgen_build_options INTERFACE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
    target_compile_options(upwgen_build_options INTERFACE
        -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Wformat=2 -Werror
        -fstack-protector-strong)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(upwgen_build_options INTERFACE -Wl,-z,relro,-z,now)
    endif()
endif()
if(UPWGEN_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "UPWGEN_SANITIZERS requires GCC or Clang")
    endif()
    target_compile_options(upwgen_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(upwgen_build_options INTERFACE -fsanitize=address,undefined)
endif()

add_library(upwgen_core STATIC src/core.cpp src/options.cpp)
target_include_directories(upwgen_core PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_BINARY_DIR}/include")
target_link_libraries(upwgen_core PRIVATE upwgen_build_options)
if(WIN32)
    target_link_libraries(upwgen_core PRIVATE bcrypt)
endif()
add_executable(upwgen src/main.cpp)
target_link_libraries(upwgen PRIVATE upwgen_core upwgen_build_options)

if(BUILD_TESTING)
    add_executable(upwgen_native_tests tests/native_tests.cpp)
    target_link_libraries(upwgen_native_tests PRIVATE upwgen_core upwgen_build_options)
    add_test(NAME native COMMAND upwgen_native_tests)
    add_test(NAME cli_version COMMAND upwgen --version)
    set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "upwgen ${PROJECT_VERSION}")
    add_test(NAME cli_default COMMAND upwgen)
    add_test(NAME cli_invalid_length COMMAND upwgen --length 2junk)
    set_tests_properties(cli_invalid_length PROPERTIES WILL_FAIL TRUE)
    add_test(NAME cli_forbidden_seed COMMAND upwgen --seed 42)
    set_tests_properties(cli_forbidden_seed PROPERTIES WILL_FAIL TRUE)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py")
        add_test(NAME cli_integration
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
                    "$<TARGET_FILE:upwgen>")
    endif()
endif()

install(TARGETS upwgen RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES LICENSE README.md DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES data/unicode/16.0.0/LICENSE.txt DESTINATION "${CMAKE_INSTALL_DOCDIR}"
        RENAME UNICODE-LICENSE.txt)
install(FILES docs/upwgen.1 DESTINATION "${CMAKE_INSTALL_MANDIR}/man1")

set(CPACK_PACKAGE_NAME upwgen)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Jon Hood")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "${PROJECT_DESCRIPTION}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR DEB)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Jon Hood")
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")
set(CPACK_PACKAGE_FILE_NAME "upwgen-${PROJECT_VERSION}-linux-amd64")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
include(CPack)
